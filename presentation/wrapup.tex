% ============================================================ Variables ===
\begin{frame}{Variables pr\'edictives}
  Six variables, identiques pour tous les mod\`eles d'accessibilit\'e :
  \vspace{0.2cm}
  \small
  \begin{itemize}
    \setlength\itemsep{0.55em}
    \item \textbf{Distance \`a la route} (OSM, routes pav\'ees) -- proximit\'e du
      r\'eseau routier praticable
    \item \textbf{Pente} (Copernicus GLO-30, $\sim$30\,m) -- difficult\'e physique de
      la derni\`ere approche
    \item \textbf{Rugosit\'e} (d\'eriv\'ee du m\^eme DEM) -- irr\'egularit\'e locale du
      relief autour du site
    \item \textbf{\'El\'evation} (Copernicus GLO-30) -- altitude, corr\'el\'ee au climat
      et \`a l'effort d'acc\`es
    \item \textbf{Friction d'occupation du sol} (ESA WorldCover) -- v\'eg\'etation dense,
      eau ou terrain b\^ati ralentissent le d\'eplacement
    \item \textbf{Distance \`a la localit\'e la plus proche} (haversine, 55 villes de
      r\'ef\'erence) -- \'eloignement des infrastructures et services
  \end{itemize}
  \vfill
  \source{M\^emes six variables pour les 733 sites \'etiquet\'es ; le mod\`ele de
  favorabilit\'e (diapositive R\'esultats) utilise un jeu diff\'erent, orient\'e
  g\'eologie/sol plut\^ot qu'infrastructure.}
\end{frame}

% ============================================================ Regle =======
\begin{frame}{R\`egle de d\'ecision \`a trois classes}
  \begin{itemize}
    \setlength\itemsep{0.9em}
    \item \textbf{Aucun mod\`ele \`a trois classes n'est jamais entra\^in\'e.} Chaque
      r\'esultat provient de l'un de deux classificateurs binaires ind\'ependants,
      chacun entra\^in\'e et valid\'e s\'epar\'ement contre sa propre v\'erit\'e terrain :
      \emph{Difficile-vs-non} et \emph{Facile-vs-non}
    \item L'\'ecran op\'erationnel \`a trois classes est une combinaison
      \textbf{d\'eterministe}, pas un troisi\`eme mod\`ele entra\^in\'e :
      \begin{itemize}
        \item un site est \textbf{Difficile} si le classificateur Difficile-vs-non le
          signale
        \item \textbf{Facile} si le classificateur Facile-vs-non le signale et que
          Difficile ne l'a pas signal\'e
        \item \textbf{Mod\'er\'e} sinon -- c'est-\`a-dire l'intersection de
          \emph{non-Difficile} et \emph{non-Facile}
      \end{itemize}
    \item Chaque pourcentage de pr\'ecision de cet expos\'e est donc une
      \textbf{exactitude binaire} contre une v\'erit\'e terrain r\'eelle -- aucun chiffre
      ne d\'ecrit directement la combinaison \`a trois classes
  \end{itemize}
  \vfill
  \source{Section \guillemotleft\,Three-Way Decision Rule\,\guillemotright\ du rapport.}
\end{frame}

% ============================================================ Modeles =====
\begin{frame}{Mod\`eles test\'es et mod\`ele retenu}
  \begin{itemize}
    \setlength\itemsep{0.8em}
    \item \textbf{Cinq familles de mod\`eles compar\'ees} sous le m\^eme protocole :
      r\'egression logistique, $k$-NN (distance g\'eographique), $k$-NN (distance dans
      l'espace des variables), Processus Gaussien, ensemble d'arbres
    \item Aucune ne d\'epasse l'ensemble d'arbres de mani\`ere significative --
      confirme un \textbf{plafond li\'e aux variables disponibles}, pas un probl\`eme de
      choix d'algorithme
    \item \textbf{Mod\`ele retenu} : ensemble \`a vote souple \`a quatre membres
      (Random Forest + XGBoost + Gradient Boosting + LightGBM), pond\'er\'e par
      confiance de l'\'etiquette et par \'equilibre des classes
    \item Configuration retenue une fois par recherche en grille sur l'ensemble des
      donn\'ees group\'ees, puis r\'eutilis\'ee pour chaque pli de validation --
      simplification courante en pratique, disclos\'ee comme limite (voir fin d'expos\'e)
  \end{itemize}
\end{frame}

% ============================================================ Validation ==
\begin{frame}{Protocole de validation}
  \centering
  \includegraphics[width=0.52\linewidth]{../report/figures/validation_schematic.pdf}
  \vspace{0.05cm}
  \footnotesize
  \begin{itemize}
    \setlength\itemsep{0.2em}
    \item \textbf{LOGO-cluster (500\,m)} : les sites tr\`es proches (quasi-doublons) sont
      group\'es et retir\'es ensemble \`a chaque pli -- \'evite la fuite de donn\'ees qui
      gonflerait artificiellement l'exactitude d'un partage al\'eatoire naïf
    \item \textbf{Leave-region-out} : entra\^in\'e sur dix r\'egions, test\'e sur la
      onzi\`eme enti\`erement retenue -- g\'en\'eralisation plus stricte, utilis\'ee pour
      Eddakhla et pour l'analyse des limites
    \item \textbf{CV par blocs spatiaux} (0,5$^\circ$$\times$0,5$^\circ$) : utilis\'ee
      uniquement pour le mod\`ele de favorabilit\'e des g\'eosites
  \end{itemize}
  \vfill
  \source{figures/validation\_schematic.pdf ; Section \guillemotleft\,Validation Strategy\,\guillemotright\ du rapport.}
\end{frame}

% ============================================================ Nouveaux ====
\begin{frame}{Nouveau : donner au mod\`ele une notion de r\'egion, prudemment}
  \footnotesize
  \begin{itemize}
    \setlength\itemsep{0.55em}
    \item Le mod\`ele national ci-dessus ne re\c{c}oit \textbf{aucune information de
      r\'egion}, pour rester utilisable ailleurs. Test compl\'ementaire : lui donner
      la r\'egion, mais de fa\c{c}on statistiquement prudente -- chaque r\'egion tire
      la pr\'ediction vers sa propre tendance, sans lui laisser une libert\'e totale
      (\'evite qu'une petite r\'egion \guillemotleft\,trompe\,\guillemotright\ le mod\`ele)
    \item Pour la classe Difficile : \textbf{77,5\,\%}, et cette fois le test
      statistique \textbf{confirme} que ce n'est pas d\^u au hasard -- l\`a o\`u le
      mod\`ele principal ne pouvait pas l'affirmer (diapositive pr\'ec\'edente)
    \item Compar\'e directement au mod\`ele principal (mêmes 733 sites) : l'avantage
      va dans le bon sens, mais reste \`a la limite de ce qu'on peut affirmer avec
      confiance -- pas encore un r\'esultat clairement tranch\'e
    \item Tient m\^eme quand on teste sur une r\'egion \textbf{compl\`etement absente
      de l'entra\^inement} (72,8\,\% contre 66,8\,\% pour le mod\`ele principal) --
      sauf \`a Eddakhla sp\'ecifiquement, o\`u le mod\`ele principal reste meilleur
      (96,0\,\% contre 92,0\,\%)
    \item \textbf{Aucun gain} \'equivalent pour la classe Facile
  \end{itemize}
  \vfill
  \source{Test de McNemar : $\chi^2=15,89$, $p=0,0001$ (vs.\ base) ; $\chi^2=3,38$,
  $p=0,066$ (vs.\ mod\`ele principal). results/json/other/mcnemar\_batch\_results.json.}
\end{frame}

\begin{frame}{Nouveau : \`a quel point peut-on faire confiance \`a une pr\'ediction ?}
  \footnotesize
  \begin{itemize}
    \setlength\itemsep{0.6em}
    \item Un pourcentage d'exactitude global ne dit pas, \textbf{site par site}, \`a
      quel point le mod\`ele est s\^ur de sa r\'eponse. Nouveau : le mod\`ele donne
      aussi une probabilit\'e de confiance pour chaque site, et on peut v\'erifier
      si ces probabilit\'es sont fiables
    \item \textbf{Premi\`ere v\'erification} : quand le mod\`ele annonce, par exemple,
      \guillemotleft\,80\,\% de chances d'\^etre Difficile\,\guillemotright, ce chiffre
      est-il r\'ealiste ? R\'eponse : d'assez bonne qualit\'e, et encore am\'elior\'e
      par un ajustement statistique simple
    \item \textbf{Deuxi\`eme v\'erification, plus utile en pratique} : \`a quelle
      fr\'equence peut-on faire confiance \`a un appel du mod\`ele \`a une seule
      classe ? Test\'e honn\^etement sur des sites jamais utilis\'es pour ce r\'eglage
    \item R\'esultat : le mod\`ele donne un \textbf{appel confiant \`a une seule classe
      pour environ 56--57\,\%} des sites, et dit honn\^etement
      \guillemotleft\,\c{c}a pourrait \^etre l'une ou l'autre\,\guillemotright\ pour
      le reste -- plut\^ot que de forcer une r\'eponse dans les cas ambigus
  \end{itemize}
  \vfill
  \source{results/json/other/calibration\_conformal\_results.json.}
\end{frame}

% ============================================================ LRO limits ==
\begin{frame}{Limite 1 : g\'en\'eralisation inter-r\'egions}
  \centering
  \includegraphics[height=2.7cm]{../report/figures/leave_region_out.pdf}

  \vspace{0.1cm}
  \scriptsize
  \begin{itemize}
    \setlength\itemsep{0.12em}
    \item \textbf{Eddakhla} (+16,0\,pts) : relief plat et homog\`ene, un seul facteur
      domine (distance \`a la route) -- terrain le plus facile \`a g\'en\'eraliser
    \item \textbf{F\'es-Mekn\'es} (+5,0\,pts) : \'echantillon le plus large (N=322),
      ce qui favorise la g\'en\'eralisation
    \item \textbf{B\'eni Mellal-Kh\'enifra} (-19,1\,pts) et \textbf{Dr\^aa-Tafilalet}
      (-6,9\,pts) : trop peu de sites \guillemotleft\,Difficile\,\guillemotright\
      (16 et 20) pour un mod\`ele fiable ; BMK combine en plus deux reliefs tr\`es
      diff\'erents (Haut Atlas et plaine du Tadla)
    \item \textbf{Tanger-T\'etouan-Al Hoceima} (-9,6), \textbf{Souss-Massa} (-10,0),
      \textbf{Rabat-Sal\'e-K\'enitra} (-19,0) : \'echantillons trop petits
      (21 \`a 52 sites) pour apprendre un motif transf\'erable
  \end{itemize}
  \vspace{0.1cm}
  \scriptsize En pratique, cette limite touche surtout l'\'evaluation : chaque
  r\'egion du catalogue actuel poss\`ede d\'ej\`a ses propres donn\'ees
  d'entra\^inement, donc les pr\'edictions r\'eelles se font toujours dans le cadre

\begin{frame}{Limite 2 : raret\'e des donn\'ees pour la classe Difficile}
  \begin{itemize}
    \setlength\itemsep{0.9em}
    \item B\'eni Mellal-Kh\'enifra (n=16) et Dr\^aa-Tafilalet (n=20) n'ont pas assez
      d'exemples \textbf{positifs} Difficile pour entra\^iner un classificateur
      r\'egional fiable
    \item Plusieurs correctifs test\'es sans succ\`es : correction du d\'es\'equilibre de
      classes, ajustement du seuil de d\'ecision, plusieurs familles de mod\`eles
      suppl\'ementaires -- aucun ne d\'epasse la base triviale de mani\`ere fiable pour
      ces deux cellules r\'egion/classe
    \item Direction la plus efficace pour la suite : \textbf{augmenter l'\'etiquetage
      cibl\'e} sur ces deux combinaisons pr\'ecises, plut\^ot qu'\'elargir le catalogue
      uniform\'ement
  \end{itemize}
\end{frame}

\begin{frame}{Limite 3 : proximit\'e routi\`ere $\ne$ technicit\'e du terrain}
  \begin{itemize}
    \setlength\itemsep{0.9em}
    \item Une grotte karstique peut se trouver \`a quelques m\`etres d'une route pav\'ee
      tout en exigeant une entr\'ee technique sp\'ecialis\'ee -- la distance \`a la route
      et la pente divergent alors en importance selon le contexte g\'eologique
    \item Confirm\'e en comparant l'importance des variables entre sites \'etiquet\'es
      depuis des sources centr\'ees route et sites \'etiquet\'es depuis des inventaires
      karstiques/hydrologiques : la pente prend le pas sur la distance \`a la route dans
      ce second groupe
    \item Visible directement sur la carte de B\'eni Mellal-Kh\'enifra (diapositive
      r\'esultats) -- trait\'e comme un \textbf{constat scientifique r\'eel}, pas comme
      une faiblesse \`a corriger
    \item Motive de traiter l'accessibilit\'e comme une propri\'et\'e vraiment
      multi-facteurs du terrain, pas une simple fonction de la proximit\'e routi\`ere
  \end{itemize}
\end{frame}
